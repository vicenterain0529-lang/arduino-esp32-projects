  MQ2CalibrationData data;

  data.baseline = mq2Baseline;
  EEPROM.put(MQ2_EEPROM_ADDR, data);
}

bool loadMQ2Baseline() {
  MQ2CalibrationData data;
  EEPROM.get(MQ2_EEPROM_ADDR, data);
  if (data.magic != MQ2_EEPROM_MAGIC) {
    return false;
  }
  if (data.baseline < 0 || data.baseline > 1023) {
    return false;
  }
  mq2Baseline = data.baseline;
  return true;
}

  MQ2CalibrationData data;
  data.magic = 0;
  data.baseline = 0;
  EEPROM.put(MQ2_EEPROM_ADDR, data);
}

float readUltrasonicRaw(byte trigPin, byte echoPin) {
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  unsigned long dur = pulseIn(echoPin, HIGH, US_TIMEOUT_US);
  if (dur == 0) {
    return 999.0f;
  }
  return dur * 0.0343f * 0.5f;
}

float readUltrasonicSensor(byte trigPin, byte echoPin) {
  float best = 999.0f;
  for (byte i = 0; i < US_SAMPLES; i++) {
    float sample = readUltrasonicRaw(trigPin, echoPin);
    if (sample < best) {
      best = sample;
    }
    delay(5);
  }
  return best;
}

float readFrontUltrasonic() {
  sFrontDist = readUltrasonicSensor(PIN_US_FRONT_TRIG, PIN_US_FRONT_ECHO);
  return sFrontDist;

void readDHT() {
  unsigned long now = millis();
  if (now - lastDhtTime < DHT_INTERVAL_MS) {
    return;
  }

  float h = dht.readHumidity();
  float t = dht.readTemperature();
  if (!isnan(h)) {
    sHumid = h;
  }
  if (!isnan(t)) {
    sTemp = t;
  }
  lastDhtTime = now;
}

void evaluateAlerts() {
  bool flameNow = flameDetectedNow();
  if (flameNow && !alertFlame) {
    alertFlame = true;
    DBGLN(F("!!! ALERT: FLAME DETECTED !!!"));
    DBG(F("A1=")); DBG(sFlameA1);
    DBG(F(" A2=")); DBG(sFlameA2);
    DBG(F(" A3=")); DBG(sFlameA3);
    DBG(F(" D1=")); DBG(sFlameD1);
    DBG(F(" D2=")); DBGLN(sFlameD2);
  } else if (!flameNow && alertFlame) {
    alertFlame = false;
    DBGLN(F("ALERT: Flame cleared"));
  }

  bool gasNow = (gasDelta() > GAS_THRESHOLD);
  if (gasNow && !alertGas) {
    alertGas = true;
    DBG(F("!!! ALERT: GAS DETECTED raw="));
    DBG(sGas);
    DBG(F(" delta="));
    DBGLN(gasDelta());
  } else if (!gasNow && alertGas) {
    alertGas = false;
    DBGLN(F("ALERT: Gas cleared"));
  }

  bool frontNow = (sFrontDist > 0.0f && sFrontDist < DIST_STOP);
  bool rearNow = (sRearDist > 0.0f && sRearDist < DIST_STOP);
  if (frontNow && !alertObstacleFront) {
    alertObstacleFront = true;
    DBG(F("!!! ALERT: FRONT OBSTACLE at "));
    DBG(sFrontDist);
    DBGLN(F("cm"));
  } else if (!frontNow && alertObstacleFront) {
    alertObstacleFront = false;
    DBGLN(F("ALERT: Front obstacle cleared"));
  }

  if (rearNow && !alertObstacleRear) {
    alertObstacleRear = true;
    DBG(F("!!! ALERT: REAR OBSTACLE at "));
    DBG(sRearDist);
    DBGLN(F("cm"));
  } else if (!rearNow && alertObstacleRear) {
    alertObstacleRear = false;
    DBGLN(F("ALERT: Rear obstacle cleared"));
  }

  bool motionNow = (sPIR1 == HIGH || sPIR2 == HIGH);
  if (motionNow && !alertMotion) {
    alertMotion = true;
    DBG(F("!!! ALERT: MOTION PIR1="));
    DBG(sPIR1);
    DBG(F(" PIR2="));
    DBGLN(sPIR2);
  } else if (!motionNow && alertMotion) {
    alertMotion = false;
  }

  bool tempNow = (sTemp > 0.0f && sTemp > TEMP_WARN);
  if (tempNow && !alertTemp) {
    alertTemp = true;
    DBG(F("!!! ALERT: HIGH TEMP "));
    DBG(sTemp);
    DBGLN(F("C"));
  } else if (!tempNow && alertTemp) {
    alertTemp = false;
    DBGLN(F("ALERT: Temp normal"));
  }
}

// ============================================================
//  AUTO SAFETY
// ============================================================

void checkAutoSafety() {
  if (!autoSafety) {
    return;
  }

  bool critical = alertFlame || (gasDelta() > GAS_DANGER) || (sTemp > TEMP_DANGER);

  if (critical) {
    if (alertFlame) {
      enterHazard(F("AUTO-SAFETY FIRE"));
    } else if (gasDelta() > GAS_DANGER) {
      enterHazard(F("AUTO-SAFETY GAS"));
    } else {
      enterHazard(F("AUTO-SAFETY HEAT"));
    }
    return;
  }

  if (hazardActive && !alertFlame && !alertGas && sTemp <= TEMP_WARN) {
    DBGLN(F("AUTO-SAFETY: Hazard cleared"));
    clearHazardState();
  }
}

char decideBestTurn(float leftDist, float rightDist) {
  if (leftDist >= DIST_CLEAR && leftDist >= rightDist) {
    return 'L';
  }
  if (rightDist >= DIST_CLEAR && rightDist > leftDist) {
    return 'R';
  }

  if (leftDist == rightDist) {
    if (lastTurnLeft) {
      return 'R';
    }
    return 'L';
  }

  return (leftDist > rightDist) ? 'L' : 'R';
}

void startScanSequence() {
  scanLeftDist = 999.0f;
  scanRightDist = 999.0f;
  scanPhase = SCAN_WAIT_LEFT;
  autoTimer = millis();
  sendCmd('<');
}

bool processScanSequence() {
  switch (scanPhase) {
    case SCAN_WAIT_LEFT:
      if (millis() - autoTimer >= AUTO_SCAN_PAUSE) {
        scanLeftDist = readFrontUltrasonic();
        sendCmd('>');
        autoTimer = millis();
        scanPhase = SCAN_WAIT_RIGHT;
      }
      return false;

    case SCAN_WAIT_RIGHT:
      if (millis() - autoTimer >= AUTO_SCAN_RIGHT_PAUSE) {
        scanRightDist = readFrontUltrasonic();
        sendCmd('C');
        autoTimer = millis();
        scanPhase = SCAN_WAIT_CENTER;
      }
      return false;

    case SCAN_WAIT_CENTER:
      if (millis() - autoTimer >= AUTO_CENTER_PAUSE) {
        pendingTurnCmd = decideBestTurn(scanLeftDist, scanRightDist);
        scanPhase = SCAN_IDLE;
        DBG(F("AUTO: Scan L="));
        DBG(scanLeftDist);
        DBG(F("cm R="));
        DBG(scanRightDist);
        DBGLN(F("cm"));
        return true;
      }
      return false;

    default:
      return false;
  }
}

void runAutonomous() {
  if (!autonomous || hazardActive) {
    return;
  }

  if (alertFlame || alertGas) {
    enterHazard(alertFlame ? F("AUTO FIRE/GAS") : F("AUTO GAS"));
    return;
  }

  switch (autoState) {
    case AUTO_PATROL:
      setMode("AUTO-PTRL");

      if (alertMotion) {
        sendCmdSilent('S');
        motionPauseUntil = millis() + AUTO_MOTION_PAUSE_MS;
        autoState = AUTO_SCANNING;
        setMode("AUTO-MOTION");
        DBGLN(F("AUTO: Motion detected, pausing"));
        break;
      }

      if (sFrontDist > 0.0f && sFrontDist < DIST_STOP) {
        sendCmdSilent('S');
        autoState = AUTO_OBSTACLE_FOUND;
        autoTimer = millis();
        DBG(F("AUTO: Obstacle at "));
        DBG(sFrontDist);
        DBGLN(F("cm"));
        break;
      }

      if (sFrontDist > 0.0f && sFrontDist < DIST_WARN) {
        setMode("AUTO-SLOW");
        sendCmd('1');
        sendCmdSilent('F');
      } else {
        sendCmd((char)('0' + AUTO_PATROL_SPEED));
        sendCmdSilent('F');
      }
      break;

    case AUTO_OBSTACLE_FOUND:
      setMode("AUTO-OBST");
      sendCmdSilent('B');
      autoTimer = millis();
      autoState = AUTO_BACKING;
      DBGLN(F("AUTO: Backing up"));
      break;

    case AUTO_BACKING:
      setMode("AUTO-BACK");
      readRearUltrasonic();
      if (sRearDist > 0.0f && sRearDist < DIST_STOP) {
        sendCmdSilent('S');
        autoState = AUTO_SCANNING;
        scanPhase = SCAN_IDLE;
        DBGLN(F("AUTO: Rear blocked, stop backing"));
      } else if (millis() - autoTimer >= AUTO_BACKUP_MS) {
        sendCmdSilent('S');
        autoState = AUTO_SCANNING;
        scanPhase = SCAN_IDLE;
        DBGLN(F("AUTO: Scanning path"));
      }
      break;

    case AUTO_SCANNING:
      if (motionPauseUntil != 0 && millis() < motionPauseUntil) {
        break;
      }
      motionPauseUntil = 0;
      setMode("AUTO-SCAN");
      if (scanPhase == SCAN_IDLE) {
        startScanSequence();
      } else if (processScanSequence()) {
        sendCmdSilent(pendingTurnCmd);
        lastTurnLeft = (pendingTurnCmd == 'L');
        autoTimer = millis();
        autoState = AUTO_TURNING;
        DBG(F("AUTO: Turning "));
        DBGLN(pendingTurnCmd == 'L' ? F("LEFT") : F("RIGHT"));
      }
      break;

    case AUTO_TURNING:
      setMode("AUTO-TURN");
      if (millis() - autoTimer >= AUTO_TURN_MS) {
        sendCmdSilent('S');
        autoState = AUTO_PATROL;
        DBGLN(F("AUTO: Resume patrol"));
      }
      break;

    case AUTO_HAZARD:
      break;

    default:
      autoState = AUTO_PATROL;
      break;
  }
}

void processInput(Stream &port, const __FlashStringHelper *label) {
  static unsigned long lastCmdTimeUSB = 0;
  static unsigned long lastCmdTimeBT = 0;
  unsigned long *lastCmdTime = (&port == &Serial) ? &lastCmdTimeUSB : &lastCmdTimeBT;

  while (port.available() > 0) {
    char c = (char)port.read();

    if (c == '\r' || c == '\n' || c == ' ') {
      continue;
    }

    unsigned long now = millis();
    if (now - *lastCmdTime < COMMAND_GAP_MS) {
      continue;
    }
    *lastCmdTime = now;

    if (&port == &Serial) {
      usbConsoleSeen = true;
      btConsoleFallback = false;
    }

    DBG(label);
    DBG(c);
    DBGNL();
    handleCommand(c);
  }
}

void updateConsoleRouting() {
  if (ENABLE_BT_DEBUG) {
    return;
  }

  if (!usbConsoleSeen && !btConsoleFallback && millis() - bootTime >= USB_FALLBACK_TO_BT_MS) {
    btConsoleFallback = true;
  }

  if (btConsoleFallback && !btFallbackAnnounced) {
    btFallbackAnnounced = true;
    BT.println(F("BT fallback active"));
    BT.println(F("No USB activity detected. Bluetooth console enabled."));
  }
}

  DBGLN(F("I2C boot scan skipped. Send 'I' to scan manually."));

  readAllSensors();

void loop() {
  updateConsoleRouting();
  processInput(Serial, F("USB> "));
  processInput(BT, F("BT>  "));

  if (millis() - lastSensorTime >= SENSOR_INTERVAL_MS) {
    lastSensorTime = millis();
    readAllSensors();
    evaluateAlerts();
    checkAutoSafety();
  }

  if (autonomous && !hazardActive) {
    runAutonomous();
  }

  if (telemActive && millis() - lastTelemTime >= TELEM_INTERVAL_MS) {
    lastTelemTime = millis();
    printTelemetry();
  }
}

                missing.add(Manifest.permission.BLUETOOTH_SCAN);
            }

            missing.add(Manifest.permission.ACCESS_FINE_LOCATION);
        }

    public void connect(Context context, DeviceItem item) {

            sendClassic(command);

        bleLineBuffer.setLength(0);

            classicReadThread.interrupt();
            classicReadThread = null;
        }

            } catch (IOException ignored) {
            }

        }

    private void connectClassic(BluetoothDevice device) {
        Thread connectThread = new Thread(() -> {
            try {
                postState("Connecting (classic) to " + safeName(device));
                BluetoothSocket candidate = device.createRfcommSocketToServiceRecord(SPP_UUID);
                bluetoothAdapter.cancelDiscovery();
                candidate.connect();

                classicSocket = candidate;
                classicOutputStream = candidate.getOutputStream();
                activeTransport = DeviceItem.Transport.CLASSIC;
                running = true;
                startClassicReader(candidate.getInputStream());
                postState("Connected (classic) to " + safeName(device));
            } catch (IOException ex) {
                postError("Classic connection failed: " + safeMessage(ex));
                disconnect();
            }
        }, "bt-classic-connect");

        connectThread.start();
    }

    @SuppressLint("MissingPermission")
    private void connectBle(Context context, BluetoothDevice device) {
        postState("Connecting (BLE) to " + safeName(device));
        activeTransport = DeviceItem.Transport.BLE;
        running = true;
        bleGatt = device.connectGatt(context, false, bleCallback, BluetoothDevice.TRANSPORT_LE);
    }

    private void sendClassic(char command) {
        OutputStream stream = classicOutputStream;
        if (stream == null) {
            postError("Not connected");
            return;
        }

        Thread writeThread = new Thread(() -> {
            try {
                stream.write(command);
                stream.flush();
            } catch (IOException ex) {
                postError("Send failed: " + safeMessage(ex));
                disconnect();
            }
        }, "bt-classic-write");

        writeThread.start();
    }

    @SuppressLint("MissingPermission")
    private void sendBle(char command) {
        if (bleGatt == null || bleWriteCharacteristic == null) {
            postError("BLE write characteristic not ready");
            return;
        }

        byte[] payload = new byte[]{(byte) command};
        bleWriteCharacteristic.setValue(payload);

        boolean started;
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
            started = bleGatt.writeCharacteristic(
                    bleWriteCharacteristic,
                    payload,
                    bleWriteCharacteristic.getWriteType()
            ) == BluetoothStatusCodes.SUCCESS;
        } else {
            started = bleGatt.writeCharacteristic(bleWriteCharacteristic);
        }

        if (!started) {
            postError("BLE send failed");
        }
    }

    private void startClassicReader(InputStream inputStream) {
        classicReadThread = new Thread(() -> {
            StringBuilder buffer = new StringBuilder();
            byte[] chunk = new byte[128];

            while (running && activeTransport == DeviceItem.Transport.CLASSIC) {
                try {
                    int read = inputStream.read(chunk);
                    if (read == -1) {
                        throw new IOException("Stream closed");
                    }
                    consumeIncoming(buffer, new String(chunk, 0, read, StandardCharsets.UTF_8));
                } catch (IOException ex) {
                    if (running) {
                        postError("Classic connection lost: " + safeMessage(ex));
                    }
                    disconnect();
                    break;
                }
            }
        }, "bt-classic-read");

        classicReadThread.start();
    }

    private void consumeIncoming(StringBuilder buffer, String incoming) {
        for (int i = 0; i < incoming.length(); i++) {
            char c = incoming.charAt(i);
            if (c == '\r') {
                continue;
            }
            if (c == '\n') {
                String line = buffer.toString().trim();
                buffer.setLength(0);
                if (!line.isEmpty()) {
                    postLine(line);
                }
            } else {
                buffer.append(c);
            }
        }
    }

    private void consumeBleBytes(byte[] value) {
        String incoming = new String(value, StandardCharsets.UTF_8);
        consumeIncoming(bleLineBuffer, incoming);
    }

    @SuppressLint("MissingPermission")
    private boolean configureBleCharacteristics(BluetoothGatt gatt) {
        List<BluetoothGattService> services = gatt.getServices();
        BluetoothGattCharacteristic preferredWrite = null;
        BluetoothGattCharacteristic preferredNotify = null;
        BluetoothGattCharacteristic fallbackWrite = null;
        BluetoothGattCharacteristic fallbackNotify = null;

        for (BluetoothGattService service : services) {
            for (BluetoothGattCharacteristic characteristic : service.getCharacteristics()) {
                int properties = characteristic.getProperties();
                boolean canWrite = (properties & BluetoothGattCharacteristic.PROPERTY_WRITE) != 0
                        || (properties & BluetoothGattCharacteristic.PROPERTY_WRITE_NO_RESPONSE) != 0;
                boolean canNotify = (properties & BluetoothGattCharacteristic.PROPERTY_NOTIFY) != 0
                        || (properties & BluetoothGattCharacteristic.PROPERTY_INDICATE) != 0;

                if (NUS_SERVICE_UUID.equals(service.getUuid())) {
                    if (NUS_RX_UUID.equals(characteristic.getUuid()) && canWrite) {
                        preferredWrite = characteristic;
                    }
                    if (NUS_TX_UUID.equals(characteristic.getUuid()) && canNotify) {
                        preferredNotify = characteristic;
                    }
                }

                if (fallbackWrite == null && canWrite) {
                    fallbackWrite = characteristic;
                }
                if (fallbackNotify == null && canNotify) {
                    fallbackNotify = characteristic;
                }
            }
        }

        bleWriteCharacteristic = preferredWrite != null ? preferredWrite : fallbackWrite;
        bleNotifyCharacteristic = preferredNotify != null ? preferredNotify : fallbackNotify;

        if (bleWriteCharacteristic == null || bleNotifyCharacteristic == null) {
            return false;
        }

        if ((bleWriteCharacteristic.getProperties() & BluetoothGattCharacteristic.PROPERTY_WRITE_NO_RESPONSE) != 0) {
            bleWriteCharacteristic.setWriteType(BluetoothGattCharacteristic.WRITE_TYPE_NO_RESPONSE);
        } else {
            bleWriteCharacteristic.setWriteType(BluetoothGattCharacteristic.WRITE_TYPE_DEFAULT);
        }

        gatt.setCharacteristicNotification(bleNotifyCharacteristic, true);
        BluetoothGattDescriptor descriptor = bleNotifyCharacteristic.getDescriptor(CCCD_UUID);
        if (descriptor != null) {
            descriptor.setValue(BluetoothGattDescriptor.ENABLE_NOTIFICATION_VALUE);
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
                gatt.writeDescriptor(descriptor, BluetoothGattDescriptor.ENABLE_NOTIFICATION_VALUE);
            } else {
                gatt.writeDescriptor(descriptor);
            }
        }

        return true;
    }

    private final BluetoothGattCallback bleCallback = new BluetoothGattCallback() {
        @Override
        public void onConnectionStateChange(BluetoothGatt gatt, int status, int newState) {
            if (status != BluetoothGatt.GATT_SUCCESS) {
                postError("BLE connection failed: status " + status);
                disconnect();
                return;
            }

            if (newState == BluetoothProfile.STATE_CONNECTED) {
                postState("Connected (BLE), discovering services");
                gatt.discoverServices();
            } else if (newState == BluetoothProfile.STATE_DISCONNECTED) {
                disconnect();
            }
        }

        @Override
        public void onServicesDiscovered(BluetoothGatt gatt, int status) {
            if (status != BluetoothGatt.GATT_SUCCESS) {
                postError("BLE service discovery failed: " + status);
                disconnect();
                return;
            }

            if (!configureBleCharacteristics(gatt)) {
                postError("No usable BLE read/write characteristics found");
                disconnect();
                return;
            }

            postState("Connected (BLE)");
        }

        @Override
        public void onCharacteristicChanged(BluetoothGatt gatt, BluetoothGattCharacteristic characteristic, byte[] value) {
            consumeBleBytes(value);
        }

        @Override
        public void onCharacteristicChanged(BluetoothGatt gatt, BluetoothGattCharacteristic characteristic) {
            byte[] value = characteristic.getValue();
            if (value != null) {
                consumeBleBytes(value);
            }
        }
    };

package com.sarrobot.controller;

import java.util.regex.Matcher;
import java.util.regex.Pattern;

public final class TelemetryParser {
    private static final Pattern LINE_PATTERN = Pattern.compile(
            "^\\[(.+?)\\]\\s*F:([\\d.]+)cm\\s*R:([\\d.]+)cm\\s*T:([\\d.]+)C\\s*H:([\\d.]+)%\\s*GAS:(\\d+)\\s*dG:(\\d+)\\s*FL:(\\d+)/(\\d+)/(\\d+)\\s*PIR:(\\d+)/(\\d+)\\s*ALRT:(.+?)\\s*AUT:(ON|OFF)\\s*HAZ:(YES|NO)$"
    );

    private TelemetryParser() {
    }

    public static RobotTelemetry parse(String line) {
        Matcher matcher = LINE_PATTERN.matcher(line.trim());
        if (!matcher.matches()) {
            return null;
        }

        RobotTelemetry telemetry = new RobotTelemetry();
        telemetry.mode = matcher.group(1).trim();
        telemetry.frontDistance = parseFloat(matcher.group(2));
        telemetry.rearDistance = parseFloat(matcher.group(3));
        telemetry.temperature = parseFloat(matcher.group(4));
        telemetry.humidity = parseFloat(matcher.group(5));
        telemetry.gasRaw = parseInt(matcher.group(6));
        telemetry.gasDelta = parseInt(matcher.group(7));
        telemetry.flameA1 = parseInt(matcher.group(8));
        telemetry.flameA2 = parseInt(matcher.group(9));
        telemetry.flameA3 = parseInt(matcher.group(10));
        telemetry.pir1 = parseInt(matcher.group(11));
        telemetry.pir2 = parseInt(matcher.group(12));
        telemetry.alerts = matcher.group(13).trim();
        telemetry.autonomous = "ON".equals(matcher.group(14));
        telemetry.hazard = "YES".equals(matcher.group(15));
        return telemetry;
    }

    private static float parseFloat(String value) {
        try {
            return Float.parseFloat(value);
        } catch (NumberFormatException ex) {
            return -1f;
        }
    }

    private static int parseInt(String value) {
        try {
            return Integer.parseInt(value);
        } catch (NumberFormatException ex) {
            return -1;
        }
    }
}

# SAR Robot Android App

This folder contains a native Android app project for your HC-05 robot controller.

## What it does

- Connects to a paired `HC-05` using classic Bluetooth SPP
- Shows live sensor telemetry from the master Arduino
- Sends movement, speed, servo, autonomous, hazard, and debug commands
- Displays the raw robot console log

## How to use it

1. Pair the phone with the `HC-05` in Android Bluetooth settings first.
2. Open the `android-app` folder in Android Studio.
3. Let Android Studio sync Gradle.
4. Build and install on your Android phone.

## Telemetry format

The parser expects lines like:

```text
[AUTO-PTRL] F:24cm R:78cm T:31C H:60% GAS:320 dG:18 FL:1023/1023/1023 PIR:0/0 ALRT:--- --- ------ ------ --- --- AUT:ON HAZ:NO
```

If you later change the Arduino telemetry format, update:

- `app/src/main/java/com/sarrobot/controller/TelemetryParser.java`

No APK was built locally here because Android Studio/Gradle is not installed in this workspace.
